import nltk
nltk.download('punkt')
nltk.download('stopwords')
nltk.download('wordnet')

from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
from nltk.stem import PorterStemmer, WordNetLemmatizer
import string

# -----------------------------
# INPUT (2 paragraphs as one)
# -----------------------------
text = """Natural Language Processing (NLP) is a fascinating field of Artificial Intelligence.
It helps computers understand human language and respond intelligently.
However, working with text data can be messy and complex.

Therefore, preprocessing techniques like removing stop words, punctuation,
and applying stemming or lemmatization are essential.
These steps improve the quality of data and make machine learning models more efficient."""

# -----------------------------
# 1. STOP WORD REMOVAL
# -----------------------------
tokens = word_tokenize(text.lower())
stop_words = set(stopwords.words('english'))

filtered_words = [word for word in tokens if word not in stop_words]

print("After Stop Word Removal:\n", filtered_words)

# -----------------------------
# 2. REMOVE PUNCTUATION
# -----------------------------
no_punct = [word for word in filtered_words if word not in string.punctuation]

print("\nAfter Removing Punctuation:\n", no_punct)

# -----------------------------
# 3. STEMMING
# -----------------------------
ps = PorterStemmer()
stemmed_words = [ps.stem(word) for word in no_punct]

print("\nAfter Stemming:\n", stemmed_words)

# -----------------------------
# 4. LEMMATIZATION
# -----------------------------
lemmatizer = WordNetLemmatizer()
lemmatized_words = [lemmatizer.lemmatize(word) for word in no_punct]

print("\nAfter Lemmatization:\n", lemmatized_words)

# -----------------------------
# FEATURE ENGINEERING
# -----------------------------
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer

# Convert list back to sentence
clean_text = " ".join(no_punct)

# -----------------------------
# Bag of Words
# -----------------------------
cv = CountVectorizer()
bow = cv.fit_transform([clean_text])

print("\nBag of Words Vocabulary:\n", cv.get_feature_names_out())
print("\nBag of Words Matrix:\n", bow.toarray())

# -----------------------------
# TF-IDF
# -----------------------------
tfidf = TfidfVectorizer()
tfidf_matrix = tfidf.fit_transform([clean_text])

print("\nTF-IDF Vocabulary:\n", tfidf.get_feature_names_out())
print("\nTF-IDF Matrix:\n", tfidf_matrix.toarray())

import nltk
nltk.download('punkt')
nltk.download('stopwords')
nltk.download('wordnet')

from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize, sent_tokenize
from nltk.stem import PorterStemmer, WordNetLemmatizer
import string


# Input Text + Pre-Positing 
text = """
Natural Language Processing is a field of AI. It helps computers understand human language.
Machine learning models require clean text to work well. So preprocessing becomes important.
We remove stopwords, punctuation and apply stemming or lemmatization to normalize words.
"""

# Pre-Positing 

tokens = word_tokenize(text.lower())
stop_words = set(stopwords.words("english"))

# Remove stopwords
filtered = [w for w in tokens if w not in stop_words]

# Remove punctuation
clean = [w for w in filtered if w not in string.punctuation]

# Stemming + Lemmatization
ps = PorterStemmer()
stemmed = [ps.stem(w) for w in clean]

lemm = WordNetLemmatizer()
lemmatized = [lemm.lemmatize(w) for w in clean]

print("Clean Text:", " ".join(clean))

# FEATURE ENGINEEERING 

from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer

clean_text = " ".join(clean)

# Bag of Words
cv = CountVectorizer()
bow = cv.fit_transform([clean_text])

print("BoW features:\n", cv.get_feature_names_out())
print(bow.toarray())

# TF-IDF
tfidf = TfidfVectorizer()
tfidf_matrix = tfidf.fit_transform([clean_text])

print("\nTF-IDF Features:\n", tfidf.get_feature_names_out())
print(tfidf_matrix.toarray())

# SUPERVISED TEXT CLASSIFICATION

dataset = [
    "Machine learning improves predictions",
    "Artificial intelligence is the future",
    "Dogs are friendly animals",
    "Cats love climbing",
]

labels = [1, 1, 0, 0]   # 1 = tech, 0 = animals

vectorizer = TfidfVectorizer()
X = vectorizer.fit_transform(dataset)

from sklearn.naive_bayes import MultinomialNB
clf = MultinomialNB()
clf.fit(X, labels)

test = ["AI will change the world"]
test_vec = vectorizer.transform(test)

print("\nPrediction:", clf.predict(test_vec))
